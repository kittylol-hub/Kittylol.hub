    local char = p.Character; if not char then return false end

    if be then local ko = be:FindFirstChild("K.O") or be:FindFirstChild("KO") or be:FindFirstChild("Knocked"); if ko and ko.Value then return true end end

    rp.FilterType = Enum.RaycastFilterType.Exclude; rp.IgnoreWater = true

end

local function doCamLock()
    if not Cfg.CL.Enabled or not clActive then return end
    if not clTarget or not clTarget.Parent then
        clActive = false; clTarget = nil; clLastVisible = nil; return
    end
    local p = Players:GetPlayerFromCharacter(clTarget.Parent)
    if not p or (Cfg.CL.DeathCheck and isDead(p)) or (Cfg.CL.KnockCheck and isKnocked(p)) or (Cfg.CL.TeamCheck and p.Team == LocalPlayer.Team) then
        clActive = false; clTarget = nil; clLastVisible = nil; return
    end
    -- Refresh to latest part
    local fresh = getHitPart(clTarget.Parent, Cfg.CL.HitPart)
    if fresh and fresh.Parent then clTarget = fresh end
    if Cfg.CL.WallCheck and not canSee(clTarget) then
        if clLastVisible and clLastVisible.Parent then clTarget = clLastVisible end
        return
    end
    clLastVisible = clTarget
    local pr = Cfg.CL.Pred; local vel = clTarget.AssemblyLinearVelocity
    local tPos = clTarget.Position + Vector3.new(vel.X*pr.X, vel.Y*pr.Y, 0)
    local cur = Camera.CFrame
    local tgt = CFrame.new(cur.Position, tPos)
    local aX = math.clamp(Cfg.CL.SmoothX, 0.001, 1)
    local aY = math.clamp(Cfg.CL.SmoothY, 0.001, 1)
    local avg = (aX+aY)*0.5
    local eaX = elasticOut(aX); local eaY = elasticOut(aY)
    local cL = cur.LookVector; local tL = tgt.LookVector
    local dir = Vector3.new(
        cL.X + (tL.X-cL.X)*(eaX*aX),
        cL.Y + (tL.Y-cL.Y)*(eaY*aY),
        cL.Z + (tL.Z-cL.Z)*(eaX*aX)
    )
    if dir.Magnitude <= 0 then return end
    local smoothCF = CFrame.new(cur.Position, cur.Position + dir.Unit)
    Camera.CFrame = smoothCF:Lerp(tgt, sineInOut(math.min(avg,1)) * avg)
end

local function _rfdhZeroCooldowns()
    -- Zero all existing cooldown values without killing
    pcall(function()
        for _, v in ipairs(game:GetDescendants()) do
            if v.Name == "ShootingCooldown" and v:IsA("ValueBase") then v.Value = 0 end
            if v.Name == "ToleranceCooldown" and v:IsA("ValueBase") then v.Value = 0 end
        end
    end)
end

local function enableRFDH()
    _rfdhZeroCooldowns()
    -- Kill self one time only 
    if not _rfdhDidKill then
        _rfdhDidKill = true
        task.spawn(function()
            task.wait(0.1)
            pcall(function() LocalPlayer.Character.Humanoid.Health = 0 end)
        end)
    end
    -- Connect to zero new cooldown instances as they appear
    if not _rfdhConn then
        _rfdhConn = game.DescendantAdded:Connect(function(v)
            if not Cfg.RFDH.Enabled then
                _rfdhConn:Disconnect(); _rfdhConn = nil; return
            end
            if v.Name == "ShootingCooldown" and v:IsA("ValueBase") then v.Value = 0 end
            if v.Name == "ToleranceCooldown" and v:IsA("ValueBase") then v.Value = 0 end
        end)
    end
end

    if not targetPart then fhDahoodTarget = nil; return end
    if Cfg.FH.FFCheck and not Cfg.FH.PrefireFF then
        if character:FindFirstChildOfClass("ForceField") then fhDahoodTarget = nil; return end
    end
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    if (root.Position - targetPart.Position).Magnitude > Cfg.FH.MaxDistance then return end
    -- redirect
    fhDahoodTarget = targetPlayer
    -- Server-side: UpdateMousePosI tells DH server where our mouse is pointing
    pcall(function() MainEvent:FireServer("UpdateMousePosI", targetPart.Position) end)
    Cfg.FH.LastShotTime = tick()
end

end

    POSITION_SWITCH_INTERVAL=0.1, STOMP_HEIGHT=4,

local function resolverTryEquipKnife()
    for _,tool in ipairs(LocalPlayer.Backpack:GetChildren()) do
        if tool:IsA("Tool") and tool.Name:lower():find("knife") then
            local hum=LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum:EquipTool(tool); return true end
        end
    end
    return false
end
local function resolverTryStab()
    if tick()-Resolver.State.lastStabTime < RES_CFG.STAB_COOLDOWN then return end
    local char=LocalPlayer.Character; if not char then return end
    for _,tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") and tool.Name:lower():find("knife") then
            pcall(function() tool:Activate() end)
            Resolver.State.lastStabTime=tick(); return
        end
    end
    resolverTryEquipKnife()
end
local function resolverSetVoidMode(enable)
    local char=LocalPlayer.Character; if not char then return end
    if enable then
        for _,part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.Name~="HumanoidRootPart" then
                Resolver.State.storedProps[part]={CanCollide=part.CanCollide,CanQuery=part.CanQuery,Transparency=part.Transparency,Material=part.Material}
                part.CanCollide=false; part.CanQuery=false; part.Transparency=1; part.Material=Enum.Material.Air
            end
        end
        local hum=char:FindFirstChildOfClass("Humanoid")
        if hum then hum.MaxHealth=math.huge; hum.Health=hum.MaxHealth; hum.PlatformStand=true end
    else
        for part,props in pairs(Resolver.State.storedProps) do
            if part and part.Parent then
                part.CanCollide=props.CanCollide; part.CanQuery=props.CanQuery
                part.Transparency=props.Transparency; part.Material=props.Material
            end
        end
        Resolver.State.storedProps={}
        local hum=char:FindFirstChildOfClass("Humanoid")
        if hum then hum.MaxHealth=100; hum.PlatformStand=false end
    end
end
local function resolverRandomDir()
    local a=math.random()*6.2831853; return Vector3.new(math.cos(a),0,math.sin(a))
end
-- SkyDive loop
local function resolverSkyDiveLoop()
    while Resolver.Active and Resolver.State.skyDiveActive do
        if Resolver.State.stomping then Resolver.State.skyDiveActive=false; break end
        if not Resolver.Target or not Resolver.Target.Character or not Resolver.Target.Character:FindFirstChild("HumanoidRootPart") then
            Resolver.State.skyDiveActive=false; break
        end
        local char=LocalPlayer.Character; if not char then break end
        local hrp=char:FindFirstChild("HumanoidRootPart"); if not hrp then break end
        Resolver.State.attached=false
        local voidCF=CFrame.new(hrp.Position.X, RES_CFG.VOID_HEIGHT, hrp.Position.Z)
        hrp.CFrame=voidCF; hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
        resolverSetVoidMode(true)
        local s=tick()
        while tick()-s < RES_CFG.SKY_TIME and Resolver.Active and Resolver.State.skyDiveActive and not Resolver.State.stomping do
            hrp.CFrame=voidCF; hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
            local hum=char:FindFirstChildOfClass("Humanoid"); if hum then hum.Health=hum.MaxHealth end
            RunService.Heartbeat:Wait()
        end
        resolverSetVoidMode(false)
        local tHRP=Resolver.Target.Character:FindFirstChild("HumanoidRootPart")
        if tHRP and Resolver.Active and Resolver.State.skyDiveActive and not Resolver.State.stomping then
            Resolver.State.attached=true
            Resolver.State.currentOffsetDir=resolverRandomDir()
            local offsetPos=tHRP.Position+Resolver.State.currentOffsetDir*RES_CFG.OFFSET_DISTANCE
            hrp.CFrame=CFrame.lookAt(offsetPos, tHRP.Position)
            task.wait(RES_CFG.GROUND_TIME)
        else Resolver.State.skyDiveActive=false end
    end
    resolverSetVoidMode(false); Resolver.State.attached=false
end
-- Resolver Heartbeat
local function resolverHeartbeatFn(dt)
    if not Resolver.Active or not Resolver.Target then return end
    local char=LocalPlayer.Character; if not char then return end
    local hrp=char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
    if Resolver.State.voidFrozen then
        local vCF=CFrame.new(hrp.Position.X, RES_CFG.VOID_HEIGHT, hrp.Position.Z)
        hrp.CFrame=vCF; hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero; return
    end
    local tChar=Resolver.Target.Character; if not tChar then return end
    local be=tChar:FindFirstChild("BodyEffects")
    local isKO=be and be:FindFirstChild("K.O") and be["K.O"].Value
    local isDead_=be and be:FindFirstChild("Dead") and be["Dead"].Value
    if isKO and not isDead_ then
        if Resolver.StompEnabled and not Resolver.State.stomping then
            Resolver.State.stomping=true; Resolver.State.attached=false; Resolver.State.skyDiveActive=false; resolverSetVoidMode(false)
        end
    end
    if Resolver.State.stomping then
        pcall(function() sethiddenproperty(hrp,"NetworkOwnership",LocalPlayer) end)
        local top=tChar:FindFirstChild("UpperTorso") or tChar:FindFirstChild("Torso") or tChar:FindFirstChild("HumanoidRootPart")
        if top then
            local sp=top.Position+Vector3.new(0, RES_CFG.STOMP_HEIGHT, 0)
            hrp.CFrame=CFrame.new(sp); hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
            if tick()-Resolver.State.lastStompSpam > RES_CFG.STOMP_SPAM_INTERVAL then
                for _=1,RES_CFG.STOMP_SPAM_COUNT do pcall(function() MainEvent:FireServer("Stomp") end) end
                Resolver.State.lastStompSpam=tick()
            end
        end
        if isDead_ then
            Resolver.State.stomping=false; Resolver.State.voidFrozen=true; resolverSetVoidMode(true)
            local vCF=CFrame.new(hrp.Position.X, RES_CFG.VOID_HEIGHT, hrp.Position.Z)
            hrp.CFrame=vCF; hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
            Resolver.State.lastStompSpam=0
        end
        return
    end
    if Resolver.State.attached then
        local tHRP=tChar:FindFirstChild("HumanoidRootPart"); if not tHRP then Resolver.State.attached=false; return end
        pcall(function() sethiddenproperty(hrp,"NetworkOwnership",LocalPlayer) end)
        pcall(function() sethiddenproperty(tHRP,"NetworkOwnership",LocalPlayer) end)
        pcall(function()
            sethiddenproperty(hrp,"PhysicsRepRootPart",tHRP)
            Resolver.State.positionSwitchTimer=Resolver.State.positionSwitchTimer+dt
            if Resolver.State.positionSwitchTimer >= RES_CFG.POSITION_SWITCH_INTERVAL then
                Resolver.State.positionSwitchTimer=0; Resolver.State.currentOffsetDir=resolverRandomDir()
            end
            local offsetPos=tHRP.Position+Resolver.State.currentOffsetDir*RES_CFG.OFFSET_DISTANCE
            hrp.CFrame=CFrame.lookAt(offsetPos, tHRP.Position)
            hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
        end)
        resolverTryStab()
        if isDead_ and not Resolver.State.voidFrozen then
            Resolver.State.attached=false; Resolver.State.stomping=false; Resolver.State.voidFrozen=true; resolverSetVoidMode(true)
            local vCF=CFrame.new(hrp.Position.X, RES_CFG.VOID_HEIGHT, hrp.Position.Z)
            hrp.CFrame=vCF; hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
        end
    end
end

    local tHRP=Resolver.Target.Character and Resolver.Target.Character:FindFirstChild("HumanoidRootPart")

            Resolver.State.beam.Transparency=NumberSequence.new(0)

local function stopResolver()
    Resolver.Active=false; Resolver.Target=nil
    for _,c in ipairs(Resolver.Connections) do c:Disconnect() end; Resolver.Connections={}
    resolverSetVoidMode(false)
    if Resolver.State.beam then Resolver.State.beam:Destroy(); Resolver.State.beam=nil end
    if Resolver.State.a0 then Resolver.State.a0:Destroy() end
    if Resolver.State.a1 then Resolver.State.a1:Destroy() end
    Notify("Resolver stopped")
end
local function startResolver()
    if not ResolverEnabled then return end
    if not ResolverTarget then Notify("Select a resolver target first"); return end
    if Resolver.Active then stopResolver() end
    Resolver.Active=true; Resolver.Target=ResolverTarget; Resolver.StompEnabled=ResolverStompEnabled
    Resolver.State={attached=false,stomping=false,skyDiveActive=false,voidFrozen=false,storedProps={},lastStabTime=0,lastStompSpam=0,positionSwitchTimer=0,currentOffsetDir=Vector3.new(0,0,-1),beam=nil,a0=nil,a1=nil,hue=0}
    Resolver.State.skyDiveActive=true
    task.spawn(resolverSkyDiveLoop)
    table.insert(Resolver.Connections, RunService.Heartbeat:Connect(resolverHeartbeatFn))
    table.insert(Resolver.Connections, RunService.RenderStepped:Connect(resolverRenderFn))
    Notify("Resolver started on " .. ResolverTarget.Name)
end

local function UpdateESP()
    for p, esp in pairs(ESPObjects) do
        for _,d in pairs(esp) do d.Visible = false end
        if not Cfg.ESP.Enabled then continue end
        if p == LocalPlayer then continue end
        if not p or not p.Parent then RemoveESP(p); continue end
        local char = p.Character; if not char then continue end
        local hrp  = char:FindFirstChild("HumanoidRootPart"); if not hrp then continue end
        local head = char:FindFirstChild("Head"); if not head then continue end
        local hum  = char:FindFirstChild("Humanoid"); if not hum or hum.Health <= 0 then continue end
        if Cfg.ESP.TeamCheck and p.Team == LocalPlayer.Team then continue end
        local lchar = LocalPlayer.Character
        local lhrp  = lchar and lchar:FindFirstChild("HumanoidRootPart")
        local dist  = lhrp and (hrp.Position - lhrp.Position).Magnitude or 0
        -- distance check removed: infinite range

        local headPos, headVis = Camera:WorldToViewportPoint(head.Position)
        local hrpPos, hrpVis   = Camera:WorldToViewportPoint(hrp.Position)
        local legPos           = Camera:WorldToViewportPoint((hrp.CFrame * CFrame.new(0,-3,0)).Position)
        if not headVis or hrpPos.Z <= 0 then continue end
        -- NaN guard
        if headPos.X ~= headPos.X or hrpPos.X ~= hrpPos.X then continue end

        local height = (Vector2.new(headPos.X, headPos.Y) - Vector2.new(legPos.X, legPos.Y)).Magnitude
        if height < 5 then continue end
        local width  = height / 2
        local tl = Vector2.new(hrpPos.X - width/2, hrpPos.Y - height/2)
        local tr = Vector2.new(hrpPos.X + width/2, hrpPos.Y - height/2)
        local bl = Vector2.new(hrpPos.X - width/2, hrpPos.Y + height/2)
        local br = Vector2.new(hrpPos.X + width/2, hrpPos.Y + height/2)

        if Cfg.ESP.ShowBoxes then
            esp.boxOutline1.From=tl; esp.boxOutline1.To=tr; esp.boxOutline1.Visible=true
            esp.boxOutline2.From=tr; esp.boxOutline2.To=br; esp.boxOutline2.Visible=true
            esp.boxOutline3.From=br; esp.boxOutline3.To=bl; esp.boxOutline3.Visible=true
            esp.boxOutline4.From=bl; esp.boxOutline4.To=tl; esp.boxOutline4.Visible=true
            esp.box1.From=tl; esp.box1.To=tr; esp.box1.Color=Cfg.ESP.BoxColor; esp.box1.Thickness=Cfg.ESP.BoxThickness; esp.box1.Visible=true
            esp.box2.From=tr; esp.box2.To=br; esp.box2.Color=Cfg.ESP.BoxColor; esp.box2.Thickness=Cfg.ESP.BoxThickness; esp.box2.Visible=true
            esp.box3.From=br; esp.box3.To=bl; esp.box3.Color=Cfg.ESP.BoxColor; esp.box3.Thickness=Cfg.ESP.BoxThickness; esp.box3.Visible=true
            esp.box4.From=bl; esp.box4.To=tl; esp.box4.Color=Cfg.ESP.BoxColor; esp.box4.Thickness=Cfg.ESP.BoxThickness; esp.box4.Visible=true
        end
        if Cfg.ESP.ShowNames then
            esp.name.Text = p.Name; esp.name.Size = Cfg.ESP.TextSize; esp.name.Color = Cfg.ESP.NameColor
            esp.name.Position = Vector2.new(hrpPos.X, tl.Y - 16); esp.name.Visible = true
        end
        if Cfg.ESP.ShowDistance then
            esp.distance.Text = "[" .. math.floor(dist) .. "m]"; esp.distance.Size = Cfg.ESP.TextSize-1; esp.distance.Color = Cfg.ESP.DistanceColor
            esp.distance.Position = Vector2.new(hrpPos.X, tl.Y - 28); esp.distance.Visible = true
        end
        if Cfg.ESP.ShowHealth then
            local hp = hum.Health; local maxhp = hum.MaxHealth
            esp.healthText.Text = math.floor(hp) .. " HP"
            esp.healthText.Color = Color3.new(1 - hp/maxhp, hp/maxhp, 0)
            esp.healthText.Size = Cfg.ESP.TextSize; esp.healthText.Position = Vector2.new(hrpPos.X, bl.Y + 4); esp.healthText.Visible = true
        end
        if Cfg.ESP.ShowHealthBar then
            local hpPct = math.clamp(hum.Health/math.max(hum.MaxHealth,1), 0, 1)
            local barX = tl.X - 5
            esp.healthBarBG.From = Vector2.new(barX, tl.Y); esp.healthBarBG.To = Vector2.new(barX, bl.Y); esp.healthBarBG.Visible = true
            esp.healthBar.From = Vector2.new(barX, bl.Y); esp.healthBar.To = Vector2.new(barX, bl.Y - height*hpPct)
            esp.healthBar.Color = Color3.new(1-hpPct, hpPct, 0); esp.healthBar.Visible = true
        end
        if Cfg.ESP.ShowTracers then
            local start = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
            esp.tracer.From = start; esp.tracer.To = Vector2.new(hrpPos.X, hrpPos.Y)
            esp.tracer.Color = Cfg.ESP.TracerColor; esp.tracer.Visible = true
        end
    end
end

local function UpdateRadar()
    if not Cfg.Radar.Enabled then
        radarBack.Visible=false; radarBorder.Visible=false; radarLocal.Visible=false
        for _,d in pairs(RadarDrawings) do d.Visible=false end; return
    end
    radarBack.Visible=true; radarBorder.Visible=true; radarLocal.Visible=true
    radarBack.Position=Cfg.Radar.Position; radarBack.Radius=Cfg.Radar.Radius; radarBack.Color=Cfg.Radar.BackColor
    radarBorder.Position=Cfg.Radar.Position; radarBorder.Radius=Cfg.Radar.Radius; radarBorder.Color=Cfg.Radar.BorderColor
    radarLocal.PointA=Cfg.Radar.Position+Vector2.new(0,-6); radarLocal.PointB=Cfg.Radar.Position+Vector2.new(-3,6); radarLocal.PointC=Cfg.Radar.Position+Vector2.new(3,6); radarLocal.Color=Cfg.Radar.LocalColor
    local center = Cfg.Radar.Position
    for p, dot in pairs(RadarDrawings) do
        local char = p.Character
        local hum2 = char and char:FindFirstChild("Humanoid")
        if char and char.PrimaryPart and hum2 and hum2.Health > 0 then
            if Cfg.Radar.HealthColor then
                dot.Color = Color3.new(1 - hum2.Health/hum2.MaxHealth, hum2.Health/hum2.MaxHealth, 0)
            elseif Cfg.Radar.TeamCheck and p.Team == LocalPlayer.Team then
                dot.Color = Cfg.Radar.TeamColor
            else dot.Color = Cfg.Radar.EnemyColor end
            local rx, rz = GetRelative(char.PrimaryPart.Position)
            local np = center - Vector2.new(rx*Cfg.Radar.Scale, rz*Cfg.Radar.Scale)
            local delta = np - center
            if delta.Magnitude < Cfg.Radar.Radius - 2 then dot.Radius=3; dot.Position=np; dot.Visible=true
            else dot.Radius=2; dot.Position=center+delta.Unit*(Cfg.Radar.Radius-2); dot.Visible=true end
        else dot.Visible=false end
    end
end

    local root = char:FindFirstChild("HumanoidRootPart"); local hum = char:FindFirstChild("Humanoid")

    end
end

local fhVoidActive, fhVoidCF     = false, nil
local fhAKState, fhAKVoidCF      = "OFF", nil
-- Dahood mode
local fhDahoodTarget = nil 
local fhStompTimer                = 0
local fhStompCount, fhStompRestoreAt, fhStompSaved = 0, 0, nil
local fhOrbit2Angle               = 0
local orbitAngle                  = 0
local groundCfmActive             = false
local saAutoFireAt                = 0
local _prevAKState                = ""

local function fhAKChecksPass(t)
    if not t or not t.Character then return false end
    local hum = t.Character:FindFirstChildOfClass("Humanoid"); if not hum or hum.Health<=0 then return false end
    local be = t.Character:FindFirstChild("BodyEffects")
    if Cfg.FH.AKFFCheck and t.Character:FindFirstChildOfClass("ForceField") then return false end
    if Cfg.FH.AKDeathCheck and be and be:FindFirstChild("Dead") and be["Dead"].Value then return false end
    if Cfg.FH.AKKnockCheck and be and be:FindFirstChild("K.O") and be["K.O"].Value then return false end
    if Cfg.FH.AKTeamCheck and t.Team == LocalPlayer.Team then return false end
    return true
end

RunService.Heartbeat:Connect(function(dt)
    local char = LocalPlayer.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")

    -- same VOID_H constant
    if Cfg.Rage.Finobe.Enabled and finobeActive and hrp then
        _finobeReal = hrp.CFrame
        pcall(function() local _nan = math.huge * 0; hrp.CFrame = _finobeReal * CFrame.new(9e9, _nan, math.huge) end)
        _finobeCrashed = true
    elseif _finobeCrashed then
        _finobeCrashed = false
        if hrp and _finobeReal then pcall(function() hrp.CFrame = _finobeReal end) end
    end

    if not hrp then return end

    -- CFrame ground movement
    if Cfg.CFM.GroundEnabled and groundCfmActive and hrp then
        local fwd=(flyKeys.W and 1 or 0)-(flyKeys.S and 1 or 0)
        local str=(flyKeys.D and 1 or 0)-(flyKeys.A and 1 or 0)
        if fwd~=0 or str~=0 then
            local camCF=Camera.CFrame
            local look=Vector3.new(camCF.LookVector.X,0,camCF.LookVector.Z)
            local right=Vector3.new(camCF.RightVector.X,0,camCF.RightVector.Z)
            if look.Magnitude>0 then look=look.Unit end
            if right.Magnitude>0 then right=right.Unit end
            local dir=(look*fwd+right*str)
            if dir.Magnitude>0 then dir=dir.Unit end
            pcall(function()
                -- Move on XZ only, 
                local newPos=Vector3.new(
                    hrp.Position.X + dir.X*Cfg.CFM.GroundSpeed*dt,
                    hrp.Position.Y,
                    hrp.Position.Z + dir.Z*Cfg.CFM.GroundSpeed*dt
                )
                hrp.CFrame=CFrame.new(newPos)*(hrp.CFrame-hrp.Position)
                hrp.AssemblyLinearVelocity=Vector3.zero
                hrp.AssemblyAngularVelocity=Vector3.zero
            end)
        end
    end

    -- FH void shooter
    if fhVoidActive and Cfg.FH.VoidEnabled and Cfg.FH.ToggleEnabled and Cfg.FH.Enabled then
        fhVoidCF = fhVoidCF or CFrame.new(hrp.Position.X, VOID_H, hrp.Position.Z)
        pcall(function() hrp.CFrame=fhVoidCF; hrp.AssemblyLinearVelocity=Vector3.zero end)
        local t2 = FH_GetCurrentTarget()
        if t2 and shouldFHTarget(t2) then FH_FireActive(t2) end
    end

    -- FH stomp (E key) - teleport to target, stomp, return
    if fhStompCount > 0 and hrp then
        if fhStompRestoreAt > 0 then
            if tick() >= fhStompRestoreAt then
                if fhStompSaved then
                    pcall(function()
                        hrp.CFrame = fhStompSaved
                        hrp.AssemblyLinearVelocity = Vector3.zero
                        hrp.AssemblyAngularVelocity = Vector3.zero
                    end)
                end
                fhStompCount=0; fhStompRestoreAt=0; fhStompSaved=nil
            end
        else
            local t2 = FH_GetCurrentTarget()
            local top = t2 and t2.Character and (t2.Character:FindFirstChild("UpperTorso") or t2.Character:FindFirstChild("Torso") or t2.Character:FindFirstChild("HumanoidRootPart"))
            if top then
                pcall(function()
                    hrp.CFrame = CFrame.new(top.Position + Vector3.new(0,4,0))
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                end)
                pcall(function() MainEvent:FireServer("Stomp") end)
            end
            fhStompCount = fhStompCount - 1
            if fhStompCount <= 0 then fhStompRestoreAt = tick() + 0.35 end
        end
    end

    -- Auto Kill state machine
    if Cfg.FH.AutoKill and Cfg.FH.ToggleEnabled and Cfg.FH.Enabled then
        local t2 = FH_GetCurrentTarget()
        if not t2 or not t2.Character then
            if fhAKState ~= "OFF" then fhAKState = "OFF"; fhAKVoidCF = nil end
        else
            local be   = t2.Character:FindFirstChild("BodyEffects")
            local hum2 = t2.Character:FindFirstChildOfClass("Humanoid")
            local isKO   = be and be:FindFirstChild("K.O") and be["K.O"].Value
            local isDead_ = (not hum2 or hum2.Health<=0) or (be and be:FindFirstChild("Dead") and be["Dead"].Value)
            local hasFF  = t2.Character:FindFirstChildOfClass("ForceField") ~= nil

            local function holdAKVoid()
                local vt = Cfg.FH.AKVoidType
                pcall(function()
                    if vt == "Random Void" then
                        fhAKVoidCF = CFrame.new((math.random()*2-1)*5000, VOID_H+math.random()*500, (math.random()*2-1)*5000)
                    else
                        fhAKVoidCF = fhAKVoidCF or CFrame.new(hrp.Position.X, VOID_H, hrp.Position.Z)
                    end
                    hrp.CFrame = fhAKVoidCF; hrp.AssemblyLinearVelocity = Vector3.zero
                end)
            end

            if fhAKState == "OFF" then
                fhAKVoidCF = CFrame.new(hrp.Position.X, VOID_H, hrp.Position.Z); fhAKState = "VOID"

            elseif fhAKState == "VOID" then
                holdAKVoid()
                if isDead_ then fhAKState = "DEAD"
                elseif Cfg.FH.AKFFCheck and hasFF then  -- wait
                elseif fhAKChecksPass(t2) then
                    local tHRP2 = t2.Character:FindFirstChild("HumanoidRootPart")
                    if tHRP2 then
                        pcall(function()
                            hrp.CFrame = tHRP2.CFrame * CFrame.new(0,0,0.5)  -- closer for faster teleport
                            hrp.AssemblyLinearVelocity = Vector3.zero
                        end)
                    end
                    fhAKState = "SHOOTING"
                end

            elseif fhAKState == "SHOOTING" then
                if isDead_ then fhAKState = "DEAD"
                elseif isKO then fhStompTimer=0; fhAKState="STOMPING"
                else
                    FH_FireActive(t2); FH_FireActive(t2); FH_FireActive(t2)
                    if Cfg.FH.AKFFCheck and hasFF then fhAKState="VOID"; fhAKVoidCF=CFrame.new(hrp.Position.X,VOID_H,hrp.Position.Z) end
                end

            elseif fhAKState == "STOMPING" then
                if isDead_ then fhAKState = "DEAD"
                else
                    local top2 = t2.Character:FindFirstChild("UpperTorso") or t2.Character:FindFirstChild("Torso") or t2.Character:FindFirstChild("HumanoidRootPart")
                    if top2 then
                        pcall(function()
                            hrp.CFrame = CFrame.new(top2.Position + Vector3.new(0,4,0))
                            hrp.AssemblyLinearVelocity = Vector3.zero
                        end)
                    end
                    fhStompTimer = fhStompTimer + dt
                    if fhStompTimer >= 0.015 then
                        fhStompTimer = 0
                        for _=1,20 do pcall(function() MainEvent:FireServer("Stomp") end) end
                    end
                end

            elseif fhAKState == "DEAD" then
                holdAKVoid()
                local dead2 = be and be:FindFirstChild("Dead") and be["Dead"].Value
                if hum2 and hum2.Health > 0 and not dead2 and not isDead_ then
                    fhAKVoidCF = CFrame.new(hrp.Position.X,VOID_H,hrp.Position.Z); fhAKState = "VOID"
                end
            end

            -- AK state notifications
            if fhAKState ~= _prevAKState then
                if fhAKState == "VOID"     then Notify("Auto Kill: voiding", 1.5)
                elseif fhAKState == "SHOOTING" then Notify("Auto Kill: shooting", 1.5)
                elseif fhAKState == "STOMPING" then Notify("Auto Kill: stomping", 1.5)
                elseif fhAKState == "DEAD"     then Notify("Auto Kill: waiting respawn", 1.5) end
                _prevAKState = fhAKState
            end
        end
    end

    -- SA auto-fire
    if Cfg.SA_AutoFire and saLockedTarget and saLockedTarget.Character and char then
        local tool = char:FindFirstChildOfClass("Tool")
        if tool and tool:FindFirstChild("GunScript") then
            if tick() >= saAutoFireAt then saAutoFireAt = tick()+0.08; pcall(function() tool:Activate() end) end
        end
    end

    -- FH Teleport (T key, checkbox required)
    if Cfg.FH.Teleport.Enabled and Cfg.FH.ToggleEnabled and Cfg.FH.Enabled and not Cfg.FH.AutoKill then
        local t2 = FH_GetCurrentTarget()
        local tHRP2 = t2 and t2.Character and t2.Character:FindFirstChild("HumanoidRootPart")
        if tHRP2 and hrp then
            pcall(function()
                hrp.CFrame = tHRP2.CFrame * CFrame.new(0,0,1.5)
                hrp.AssemblyLinearVelocity = Vector3.zero
            end)
        end
    end

    -- FH Spectate
    if Cfg.FH.Spectate then
        local t2  = FH_GetCurrentTarget()
        local hum2 = t2 and t2.Character and t2.Character:FindFirstChildOfClass("Humanoid")
        if hum2 and hum2.Health > 0 then
            Camera.CameraType = Enum.CameraType.Custom; Camera.CameraSubject = hum2
        else
            local selfHum = char and char:FindFirstChildOfClass("Humanoid")
            if selfHum and Camera.CameraSubject ~= selfHum then Camera.CameraSubject = selfHum; Camera.CameraType = Enum.CameraType.Custom end
        end
    else
        local selfHum = char and char:FindFirstChildOfClass("Humanoid")
        if selfHum and Camera.CameraSubject ~= selfHum then Camera.CameraSubject = selfHum; Camera.CameraType = Enum.CameraType.Custom end
    end

    -- SA Orbit (Rage tab)
    if Cfg.Rage.Orbit.Enabled and saLockedTarget and saLockedTarget.Character then
        local tHRP = saLockedTarget.Character:FindFirstChild("HumanoidRootPart")
        if tHRP and tHRP.Parent and hrp.Parent then
            orbitAngle = (orbitAngle + dt * Cfg.Rage.Orbit.Speed) % 6.2831853
            local ox = math.cos(orbitAngle) * Cfg.Rage.Orbit.Radius
            local oz = math.sin(orbitAngle) * Cfg.Rage.Orbit.Radius
            local np = tHRP.Position + Vector3.new(ox, Cfg.Rage.Orbit.Height, oz)
            pcall(function()
                hrp.CFrame = CFrame.new(np, tHRP.Position)
                hrp.AssemblyLinearVelocity = Vector3.zero; hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    -- FH Orbit2 (two presets: 3D Chaos and Circle)
    -- DahoodAutoOrbit: runs whenever ToggleEnabled and a target exists (no fire key needed)
    local _dahoodOrbitTarget = Cfg.FH.DahoodAutoOrbit and Cfg.FH.Mode == "Dahood" and Cfg.FH.ToggleEnabled
        and (FH_GetCurrentTarget() or fhGetClosest())
    local _orbitRunning = (Cfg.FH.Orbit2.Enabled and Cfg.FH.Orbit2.Active)
        or (_dahoodOrbitTarget ~= nil and _dahoodOrbitTarget ~= false)
    local _orbitPreset = (Cfg.FH.DahoodAutoOrbit and Cfg.FH.Mode == "Dahood" and not (Cfg.FH.Orbit2.Enabled and Cfg.FH.Orbit2.Active)) and "3D Chaos" or Cfg.FH.Orbit2.Preset
    if _orbitRunning then
        local t2 = (type(_dahoodOrbitTarget) == "table" and _dahoodOrbitTarget) or FH_GetCurrentTarget()
        local tHRP2 = t2 and t2.Character and t2.Character:FindFirstChild("HumanoidRootPart")
        if tHRP2 and tHRP2.Parent and hrp.Parent then
            fhOrbit2Angle = (fhOrbit2Angle + dt * Cfg.FH.Orbit2.Speed) % 6.2831853
            local ox, oy, oz
            local preset = _orbitPreset
            if preset == "3D Chaos" then
                -- 3D Chaos: combines circular orbit with sinusoidal up/down motion
                -- Uses two independently running sine waves to make it fully chaotic
                local hAngle = fhOrbit2Angle
                local vPhase = fhOrbit2Angle * 1.7  -- different frequency for vertical
                ox = math.cos(hAngle) * Cfg.FH.Orbit2.Radius
                oz = math.sin(hAngle) * Cfg.FH.Orbit2.Radius
                -- Vertical: oscillates between -Height and +Height continuously
                oy = math.sin(vPhase) * math.abs(Cfg.FH.Orbit2.Height + 2)
                    + math.cos(fhOrbit2Angle * 2.3) * 1.5  -- secondary wobble
            else -- "Circle"
                ox = math.cos(fhOrbit2Angle) * Cfg.FH.Orbit2.Radius
                oy = Cfg.FH.Orbit2.Height
                oz = math.sin(fhOrbit2Angle) * Cfg.FH.Orbit2.Radius
            end
            pcall(function()
                hrp.CFrame = CFrame.new(tHRP2.Position + Vector3.new(ox, oy, oz), tHRP2.Position)
                hrp.AssemblyLinearVelocity = Vector3.zero; hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    -- Glue (SA or FH target)
    if Cfg.Rage.Glue.Enabled and Cfg.Rage.Glue.Active then
        local glueTgt = saLockedTarget or FH_GetCurrentTarget()
        local tHRP = glueTgt and glueTgt.Character and glueTgt.Character:FindFirstChild("HumanoidRootPart")
        if tHRP and tHRP.Parent and hrp.Parent then
            pcall(function()
                local tCF = tHRP.CFrame
                local bp  = (tCF * CFrame.new(0,0,1.4)).Position
                hrp.CFrame = CFrame.new(Vector3.new(bp.X, tHRP.Position.Y, bp.Z), tHRP.Position)
                hrp.AssemblyLinearVelocity  = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        else
            Cfg.Rage.Glue.Active = false
        end
    end
end)

local _EMOTES = {{name="Godlike",id="rbxassetid://3823158750"},{name="Floss Dance",id="rbxassetid://5917570207"},{name="Flying Parts",id="rbxassetid://134563511050560"},{name="Godly Aura",id="rbxassetid://76361248833307"},{name="Take The L",id="rbxassetid://108012578877632"}}
local _emoteTrack = nil
local function _playEmote(id)
    task.spawn(function()
        local char = LocalPlayer.Character; if not char then return end
        local hum  = char:FindFirstChildOfClass("Humanoid"); if not hum then return end
        -- Stop any current emote first
        if _emoteTrack then
            pcall(function() _emoteTrack:Stop() end)
            _emoteTrack = nil
        end
        -- Use Animator:LoadAnimation to 
        local animator = hum:FindFirstChildOfClass("Animator")
        if not animator then
            animator = Instance.new("Animator"); animator.Parent = hum
        end
        local anim = Instance.new("Animation")
        anim.AnimationId = id
        -- LoadAnimation on the Animator directly works on all executors
        local ok, track = pcall(function() return animator:LoadAnimation(anim) end)
        if not ok or not track then
            -- Fallback to hum:LoadAnimation
            ok, track = pcall(function() return hum:LoadAnimation(anim) end)
            if not ok or not track then return end
        end
        _emoteTrack = track
        _emoteTrack.Priority = Enum.AnimationPriority.Action4  -- highest priority, overrides everything
        _emoteTrack.Looped   = true
        _emoteTrack:Play(0, 1, 1)
    end)
end
local function _stopEmote() pcall(function() if _emoteTrack then _emoteTrack:Stop();_emoteTrack=nil end end) end
for _,em in ipairs(_EMOTES) do local emId=em.id; EmoteGroup:AddToggle("Emote_"..string.gsub(em.name," ",""),{Text=em.name,Default=false,Callback=function(v) for _,e2 in ipairs(_EMOTES) do if e2.id~=emId then pcall(function() if Options["Emote_"..string.gsub(e2.name," ","")] then Options["Emote_"..string.gsub(e2.name," ","")]:SetValue(false) end end) end end; if v then _playEmote(emId) else _stopEmote() end end}) end
EmoteGroup:AddButton("stop emote", _stopEmote)
